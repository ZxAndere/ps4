cmake_minimum_required(VERSION 3.16)
project(PS4FrameGen VERSION 0.3.0 LANGUAGES CXX)
option(FRAMEGEN_SANITIZE "Enable address and undefined-behavior sanitizers on host" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(framegen
  engine/capture/image.cpp engine/motion/block_match.cpp
  engine/interpolation/interpolate.cpp engine/scaling/scale.cpp
  engine/performance/pipeline.cpp engine/presentation/monitor.cpp
  engine/optimized/processor.cpp engine/performance/runtime.cpp
  overlay/input/controller.cpp overlay/settings/settings.cpp overlay/menu/draw.cpp)
target_include_directories(framegen PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(framegen PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
target_link_libraries(framegen PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(framegen PRIVATE /W4 /permissive-)
else()
  target_compile_options(framegen PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FRAMEGEN_SANITIZE AND NOT MSVC)
  target_compile_options(framegen PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(framegen PUBLIC -fsanitize=address,undefined)
endif()
add_executable(framegen_cli windows-tests/main.cpp)
target_link_libraries(framegen_cli PRIVATE framegen)
add_executable(framegen_ablation windows-tests/ablation.cpp)
target_link_libraries(framegen_ablation PRIVATE framegen)
enable_testing()
add_executable(framegen_tests tests/tests.cpp)
target_link_libraries(framegen_tests PRIVATE framegen)
add_test(NAME framegen_functional COMMAND framegen_tests)
add_library(framegen_bridge SHARED goldhen/plugin/bridge.cpp)
target_compile_definitions(framegen_bridge PRIVATE FG_BRIDGE_BUILD)
target_link_libraries(framegen_bridge PRIVATE framegen)
add_executable(framegen_bridge_tests tests/bridge_tests.cpp)
target_link_libraries(framegen_bridge_tests PRIVATE framegen_bridge framegen)
add_test(NAME framegen_cooperative_bridge COMMAND framegen_bridge_tests)

add_executable(framegen_runtime_tests tests/runtime_tests.cpp)
target_link_libraries(framegen_runtime_tests PRIVATE framegen)
add_test(NAME framegen_runtime COMMAND framegen_runtime_tests)
add_executable(framegen_inspect windows-tests/inspect.cpp)
target_link_libraries(framegen_inspect PRIVATE framegen)
add_executable(framegen_benchmark_v03 windows-tests/benchmark-v03.cpp)
target_link_libraries(framegen_benchmark_v03 PRIVATE framegen)
add_executable(framegen_replay windows-tests/replay.cpp)
target_link_libraries(framegen_replay PRIVATE framegen)
